cmake_minimum_required(VERSION 3.24)
project(kvtransfer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(KVT_CUDA "Build CUDA staging" OFF)
option(KVT_PYTHON "Build Python bindings" OFF)
set(KVT_SANITIZER "" CACHE STRING "address or thread")
if(KVT_SANITIZER STREQUAL "address")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(KVT_SANITIZER STREQUAL "thread")
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
elseif(NOT KVT_SANITIZER STREQUAL "")
  message(FATAL_ERROR "Unknown sanitizer: ${KVT_SANITIZER}")
endif()
find_package(Threads REQUIRED)
find_package(Protobuf REQUIRED)
protobuf_generate_cpp(PROTO_SRCS PROTO_HDRS proto/control.proto)
add_library(kvtransfer src/protocol.cpp src/staging.cpp src/session.cpp src/transport.cpp src/engine.cpp ${PROTO_SRCS})
target_include_directories(kvtransfer PUBLIC include ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(kvtransfer PUBLIC protobuf::libprotobuf Threads::Threads)
target_compile_options(kvtransfer PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(KVT_CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(kvtransfer PRIVATE src/cuda_staging.cpp)
  target_link_libraries(kvtransfer PUBLIC CUDA::cudart)
  target_compile_definitions(kvtransfer PUBLIC KVT_CUDA=1)
endif()
if(KVT_PYTHON)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_native src/bindings.cpp)
  target_link_libraries(_native PRIVATE kvtransfer)
  install(TARGETS _native LIBRARY DESTINATION kvtransfer)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  add_executable(kvtransfer_tests tests/core_test.cpp tests/engine_test.cpp)
  target_link_libraries(kvtransfer_tests PRIVATE kvtransfer GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(kvtransfer_tests DISCOVERY_TIMEOUT 30)
endif()
add_executable(fake_kv tools/fake_kv.cpp)
target_link_libraries(fake_kv PRIVATE kvtransfer)
